cmake_minimum_required(VERSION 3.20)
project(TcpIpTester LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
endif()
enable_testing()
add_executable(protocol_tests tests/protocol_tests.cpp)
target_include_directories(protocol_tests PRIVATE src)
add_test(NAME protocol COMMAND protocol_tests)
if(WIN32)
  add_executable(image_bridge tests/image_bridge.cpp)
  target_include_directories(image_bridge PRIVATE src)
  target_compile_definitions(image_bridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(image_bridge PRIVATE ws2_32 windowscodecs ole32 gdi32)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import PIL" RESULT_VARIABLE PIL_MISSING)
    if(NOT PIL_MISSING)
      add_test(NAME image_python_cpp COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_image_interop.py" $<TARGET_FILE:image_bridge>)
      set_tests_properties(image_python_cpp PROPERTIES TIMEOUT 90)
    endif()
  endif()
  add_executable(tcp_integration_tests tests/tcp_integration_tests.cpp)
  target_include_directories(tcp_integration_tests PRIVATE src)
  target_compile_definitions(tcp_integration_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(tcp_integration_tests PRIVATE ws2_32)
  add_test(NAME tcp_two_processes COMMAND tcp_integration_tests)
  set_tests_properties(tcp_two_processes PROPERTIES TIMEOUT 20)
  add_executable(TcpIpTester WIN32 src/main.cpp)
  # This small Win32 teaching UI deliberately uses message-based socket events.
  target_compile_definitions(TcpIpTester PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WINSOCK_DEPRECATED_NO_WARNINGS)
  target_link_libraries(TcpIpTester PRIVATE ws2_32 comdlg32 windowscodecs ole32)
  set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT TcpIpTester)
endif()
